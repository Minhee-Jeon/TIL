4. Zustand Store로 모달 관리
```tsx
// stores/ModalStore.ts
import { create } from 'zustand';

interface ModalState {
  activeModals: string[];
  openModal: (modalId: string) => void;
  closeModal: (modalId: string) => void;
  closeAllModals: () => void;
  isModalOpen: (modalId: string) => boolean;
}

export const useModalStore = create<ModalState>((set, get) => ({
  activeModals: [],
  
  openModal: (modalId: string) => {
    set(state => ({
      activeModals: [...state.activeModals, modalId]
    }));
  },
  
  closeModal: (modalId: string) => {
    set(state => ({
      activeModals: state.activeModals.filter(id => id !== modalId)
    }));
  },
  
  closeAllModals: () => {
    set({ activeModals: [] });
  },
  
  isModalOpen: (modalId: string) => {
    return get().activeModals.includes(modalId);
  },
}));
```

```tsx // utils/Setter.ts
import update, { type Spec } from "immutability-helper";
import { useMemo } from "react";

const lowerFirst = <T extends string>(str: T): Lowercase<T> =>
  (str.charAt(0).toLowerCase() + str.slice(1)) as Lowercase<T>;

export type Setters<S> = {
  [K in Extract<keyof S, string> as `set${Capitalize<K>}`]: (
    command: Spec<S[K]>,
  ) => void;
};

const createSetters = <S>(
  setState: React.Dispatch<React.SetStateAction<Partial<S>>>,
): Setters<S> =>
  useMemo(() => {
    const cache: Partial<Record<string, (...args: any[]) => void>> = {};
    return new Proxy({} as Setters<S>, {
      get(_target, prop: string) {
        if (!prop.startsWith("set")) return undefined;
        if (cache[prop]) return cache[prop];
        const key = lowerFirst(prop.slice(3)) as keyof S;
        const setter = (command: Spec<S[typeof key]>) =>
          setState((prev) => update(prev, { [key]: command } as any));
        cache[prop] = setter;
        return setter;
      },
    }) as Setters<S>;
  }, [setState]);

export const makeSliceSetters = <Slice, Root>(
  sliceKey: keyof Root & string,
  set: (fn: (state: Root) => Partial<Root>) => void,
): Setters<Slice> => {
  return new Proxy({} as Setters<Slice>, {
    get(_target, prop: string) {
      if (!prop.startsWith("set")) return undefined as any;
      const key = lowerFirst(prop.slice(3)) as keyof Slice;
      return (command: Spec<Slice[typeof key]>) => {
        set(
          (state) =>
            ({
              [sliceKey]: update((state as any)[sliceKey], {
                [key]: command,
              } as any),
            }) as Partial<Root>,
        );
      };
    },
  }) as Setters<Slice>;
};

export default createSetters;
```
